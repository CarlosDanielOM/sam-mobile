cmake_minimum_required(VERSION 3.22)
project(sam_embeddings LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp" CACHE PATH "Pinned llama.cpp checkout")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LLAMA_SOURCE}"
    OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_REV STREQUAL "465e49b9cea78a68b9c244ffb48d0ee24a82873d")
    message(FATAL_ERROR "Unexpected llama.cpp revision: ${LLAMA_REV}")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
if(ANDROID)
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory("${LLAMA_SOURCE}" llama EXCLUDE_FROM_ALL)
add_library(sam-engine STATIC src/main/cpp/engine.cpp)
target_include_directories(sam-engine PUBLIC src/main/cpp PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(sam-engine PUBLIC llama)
if(ANDROID)
    add_library(sam-embeddings SHARED src/main/cpp/jni.cpp)
    set_target_properties(sam-embeddings PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
    target_include_directories(sam-embeddings PRIVATE "${LLAMA_SOURCE}/vendor")
    target_link_libraries(sam-embeddings PRIVATE sam-engine)
    target_link_options(sam-embeddings PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
    add_executable(sam-host-smoke tests/host-smoke.cpp)
    target_include_directories(sam-host-smoke PRIVATE "${LLAMA_SOURCE}/vendor")
    target_link_libraries(sam-host-smoke PRIVATE sam-engine)
    add_executable(sam-batch-investigate tests/batch-investigate.cpp)
    target_include_directories(sam-batch-investigate PRIVATE "${LLAMA_SOURCE}/vendor")
    target_link_libraries(sam-batch-investigate PRIVATE llama)
endif()
